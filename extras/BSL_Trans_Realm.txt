# Trans Realm preset for BSL Shaders 10.1.8 (Iris)
#
# How to use:
#   1. In Minecraft, pick another shader pack (or none) first, or close the game.
#   2. Copy this file into .minecraft/shaderpacks/, next to BSL_v10.1.8.zip.
#   3. Rename it to the zip's exact name plus ".txt", replacing the old one: BSL_v10.1.8.zip.txt
#   4. Pick BSL under Video Settings > Shader Packs.
#
# Every option and value here is one BSL 10.1.8 offers (checked against its lib/settings.glsl). BSL ignores a value it
# doesn't offer: the old preset's CLOUDS=false, AURORA=true and CG_TM=0.20 were all ignored, so BSL kept its own
# clouds over the heart clouds and the pink colour grade never came on.
#
# What it does:
#   - Clouds: BSL's "Vanilla" clouds, so the realm's heart clouds show. The mod tints them with the game's cloud colour,
#     so they cycle through the flag (blue, pink, white, pink, blue) under BSL too. Minecraft's own Clouds video
#     setting must be on (Fast or Fancy).
#   - Sky and fog follow the game's: the Trans Realm keeps its trans-blue sky with a pink horizon and pink fog, and the
#     Overworld stays normal.
#   - Water takes each biome's colour: blue seas, pink rivers, lavender crystal pools.
#   - A pink and blue aurora at night. It shows in every overworld-like dimension, the Overworld too: set AURORA to 1
#     for snowy biomes only, or 0 to turn it off.
#   - Pink sunrises and sunsets, pink-white daylight, lavender nights, lilac shadows, pink torchlight.
#   - Brighter colours, a soft pink colour grade, extra bloom and glow.
#
# Lighting strengths stay at BSL's defaults. Changing anything in the shader menu rewrites this file.

# Clouds: vanilla clouds (0 off, 1 skybox, 2 volumetric, 3 vanilla)
CLOUDS=3

# Sky, fog and water from the game (follows the realm's own colours)

WATER_MODE=1

# Sky colour used where the game's isn't
SKY_R=92
SKY_G=204

# Sunlight: morning, day, evening, night
LIGHT_MR=252

LIGHT_DR=255
LIGHT_DG=232
LIGHT_DB=244
LIGHT_ER=244
LIGHT_EG=128
LIGHT_EB=184

LIGHT_NG=160
LIGHT_NB=255

AMBIENT_MR=216

AMBIENT_MB=220
AMBIENT_DR=168
AMBIENT_DG=204
AMBIENT_DB=252
AMBIENT_ER=216

AMBIENT_NR=148
AMBIENT_NG=128

# Torches and lanterns

# Water colour used where the biome's isn't
